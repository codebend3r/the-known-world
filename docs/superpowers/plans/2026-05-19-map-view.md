# Map View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a pannable, zoomable SVG map of the North as the region's primary view, with clickable castle markers that route to existing castle detail pages.

**Architecture:** A placeholder hand-traced SVG of the North sits in `public/map/north.svg`. A `MapStage` client component wraps `react-svg-pan-zoom` around it. Castle markers are rendered as React children of the SVG, positioned by their frontmatter `coords`. A parent `NorthMapView` client component manages layer-toggle state. The route `/the-north` (a server component) loads all castles at build time and hands them to `NorthMapView`.

**Tech Stack additions:** `react-svg-pan-zoom` (or a hand-rolled pan/zoom hook if the package is unmaintained; see Task 5 notes). Everything else carries over from Plan 1.

**Out of scope for this plan:**

- Real (non-placeholder) North SVG with accurate geography
- Battle-site pins (depends on event data, which arrives in a later plan)
- Sub-region shaded overlays
- Timeline view, Houses view, Family Tree view, scraper, PWA

import type { Castle } from "./schemas";

type Loaded<T> = { frontmatter: T; body: string; slug: string };
type CastleType = Castle["type"];

export const ALL_CASTLE_TYPES: CastleType[] = [
  "castle",
  "town",
  "ruin",

export function selectVisibleCastles(

  enabledTypes: Set<CastleType>,

}

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import { ReactSVGPanZoom, TOOL_AUTO, type Value } from "react-svg-pan-zoom";
import type { ReactNode } from "react";

const VIEWBOX_WIDTH = 800;
const VIEWBOX_HEIGHT = 1000;

type Props = {
  children: ReactNode;
  svgUrl: string;
};

export function MapStage({ children, svgUrl }: Props) {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const [size, setSize] = useState<{ w: number; h: number } | null>(null);
  const [value, setValue] = useState<Value>({} as Value);

  useEffect(() => {
    if (!containerRef.current) return;
    const el = containerRef.current;
    const update = () => setSize({ w: el.clientWidth, h: el.clientHeight });
    update();
    const ro = new ResizeObserver(update);
    ro.observe(el);
    return () => ro.disconnect();
  }, []);

  return (
    <div ref={containerRef} className="map-stage">
      <div className="map-stage-inner">
        {size && (
          <ReactSVGPanZoom
            width={size.w}
            height={size.h}
            tool={TOOL_AUTO}
            onChangeTool={() => {}}
            value={value}
            onChangeValue={setValue}
            background="transparent"
            SVGBackground="transparent"
            detectAutoPan={false}
            scaleFactor={1.2}
            scaleFactorMin={0.5}
            scaleFactorMax={6}
            preventPanOutside={true}
            toolbarProps={{ position: "none" }}
            miniatureProps={{
              position: "none",
              background: "transparent",
              width: 0,
              height: 0,
            }}
          >
            <svg width={VIEWBOX_WIDTH} height={VIEWBOX_HEIGHT}>
              <image
                href={svgUrl}
                x={0}
                y={0}
                width={VIEWBOX_WIDTH}
                height={VIEWBOX_HEIGHT}
              />
              {children}
            </svg>
          </ReactSVGPanZoom>
        )}
      </div>
    </div>
  );
}
```

### Fallback if `react-svg-pan-zoom` is unavailable

Replace the body of `MapStage` with a hand-rolled pan/zoom container using pointer events. The interface (props: `children`, `svgUrl`) stays identical so Task 7 doesn't change. Outline:

```tsx
"use client";
import { useRef, useState, useCallback } from "react";
import type { ReactNode, PointerEvent } from "react";

const VIEWBOX_WIDTH = 800;
const VIEWBOX_HEIGHT = 1000;

type Props = { children: ReactNode; svgUrl: string };

export function MapStage({ children, svgUrl }: Props) {
  const [scale, setScale] = useState(1);
  const [tx, setTx] = useState(0);
  const [ty, setTy] = useState(0);
  const dragging = useRef<{ x: number; y: number } | null>(null);

  const onPointerDown = (e: PointerEvent<SVGSVGElement>) => {
    (e.target as SVGSVGElement).setPointerCapture(e.pointerId);
    dragging.current = { x: e.clientX, y: e.clientY };
  };
  const onPointerMove = (e: PointerEvent<SVGSVGElement>) => {
    if (!dragging.current) return;
    setTx((t) => t + (e.clientX - dragging.current!.x));
    setTy((t) => t + (e.clientY - dragging.current!.y));
    dragging.current = { x: e.clientX, y: e.clientY };
  };
  const onPointerUp = () => {
    dragging.current = null;
  };
  const onWheel = useCallback((e: React.WheelEvent<SVGSVGElement>) => {
    e.preventDefault();
    const delta = e.deltaY < 0 ? 1.1 : 1 / 1.1;
    setScale((s) => Math.min(6, Math.max(0.5, s * delta)));
  }, []);

  return (
    <div className="map-stage">
      <svg
        className="map-stage-inner"
        viewBox={`0 0 ${VIEWBOX_WIDTH} ${VIEWBOX_HEIGHT}`}
        onPointerDown={onPointerDown}
        onPointerMove={onPointerMove}
        onPointerUp={onPointerUp}
        onPointerLeave={onPointerUp}
        onWheel={onWheel}
        style={{ touchAction: "none" }}
      >
        <g transform={`translate(${tx} ${ty}) scale(${scale})`}>
          <image
            href={svgUrl}
            x={0}
            y={0}
            width={VIEWBOX_WIDTH}
            height={VIEWBOX_HEIGHT}
          />
          {children}
        </g>
      </svg>
    </div>
  );
}
```

## Task 8: NorthMapView (stateful composition)

**Files:**

- Create: `components/NorthMapView.tsx`

A client component that holds the layer-toggle state and composes `MapStage` + `MapLayerToggle` + `MapMarker`s. Receives the full castle list from a server component parent (Task 9) and filters in-memory by enabled types.

- [ ] **Step 1: Create `components/NorthMapView.tsx`**

```tsx
"use client";

import { useState } from "react";
import type { Castle } from "@/lib/schemas";
import { ALL_CASTLE_TYPES, selectVisibleCastles } from "@/lib/map";
import { MapStage } from "./MapStage";
import { MapMarker } from "./MapMarker";
import { MapLayerToggle } from "./MapLayerToggle";

type CastleType = Castle["type"];
type Loaded<T> = { frontmatter: T; body: string; slug: string };

type Props = {
  castles: Array<Loaded<Castle>>;
};

export function NorthMapView({ castles }: Props) {
  const [enabled, setEnabled] = useState<Set<CastleType>>(
    new Set(ALL_CASTLE_TYPES),
  );

  const toggle = (type: CastleType) => {
    setEnabled((prev) => {
      const next = new Set(prev);
      if (next.has(type)) next.delete(type);
      else next.add(type);
      return next;
    });
  };

  const visible = selectVisibleCastles(castles, enabled);

  return (
    <>
      <MapLayerToggle enabled={enabled} onToggle={toggle} />
      <MapStage svgUrl="/map/north.svg">
        {visible.map((c) => (
          <MapMarker
            key={c.frontmatter.slug}
            slug={c.frontmatter.slug}
            name={c.frontmatter.name}
            type={c.frontmatter.type}
            cx={c.frontmatter.coords.x}
            cy={c.frontmatter.coords.y}
          />
        ))}
      </MapStage>
    </>
  );
}
```
